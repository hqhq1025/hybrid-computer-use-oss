// Measurement harness, not committed: replays historical AX arguments and
// writes one verdict per record. Compiled unchanged against the old and the
// new tree.

	panic(historyReached{})

var historyRef = regexp.MustCompile(`^([rwaf])[1-9][0-9]*$`)

// rewriteRefs maps every exact short ref to the fixture's r1/w1/a1/f1 so that
// ref lifecycle never decides the verdict.
func rewriteRefs(value any) any {
	switch typed := value.(type) {
	case map[string]any:
		out := map[string]any{}
		for key, item := range typed {
			out[key] = rewriteRefs(item)
		}
		return out
	case []any:
		out := make([]any, len(typed))
		for index, item := range typed {
			out[index] = rewriteRefs(item)
		}
		return out
	case string:
		if match := historyRef.FindStringSubmatch(typed); match != nil {
			return match[1] + "1"
		}
		return typed
	default:
		return value
	}
}

func historyVerdict(args map[string]any) (accepted bool, text string) {
	service, _, _, _ := sheetBatchService(&testing.T{})
	service.backend = historyBackend{&recordingPlatformBackend{err: errors.New("unreachable")}}
	snapshot := editorFormSnapshot()
	service.findCursorRegistry.Create(snapshot, snapshot.Elements, 0, "probe")
	defer func() {
		if recovered := recover(); recovered != nil {
			if _, ok := recovered.(historyReached); !ok {
				accepted, text = false, fmt.Sprintf("panic: %v", recovered)
				return
			}
			accepted, text = true, "reached provider"
		}
	}()
	result := service.callTool("AX", args)
	text = strings.SplitN(resultText(result), "\n", 2)[0]
	for _, part := range result.Content {
		if part.Type == "text" && !strings.HasPrefix(part.Text, "[ax]") {
			text = strings.SplitN(part.Text, "\n", 2)[0]
			break
		}
	}
	for _, fragment := range []string{
		"cannot use target.index without a current accessibility snapshot",
		"AX find cursor has no recoverable snapshot",
		"is not present in the fresh canonical snapshot",
	} {
		if strings.Contains(text, fragment) {
			return true, text
		}
	}
	return !result.IsError, text
}

func TestHistoryReplay(t *testing.T) {
	inputs := os.Getenv("HCU_HISTORY_INPUTS")
	output := os.Getenv("HCU_HISTORY_OUTPUT")
	if inputs == "" || output == "" {
		t.Skip("measurement harness")
	}
	t.Setenv("COMPUTER_USE_ERROR_CONTEXT", "off")
	out, err := os.Create(output)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	writer := bufio.NewWriter(out)
	defer writer.Flush()
	for _, path := range strings.Split(inputs, ",") {
		file, err := os.Open(path)
		if err != nil {
			t.Fatal(err)
		}
		reader, err := gzip.NewReader(file)
		if err != nil {
			t.Fatal(err)
		}
		scanner := bufio.NewScanner(reader)
		scanner.Buffer(make([]byte, 1<<20), 1<<24)
		line := 0
		for scanner.Scan() {
			line++
			var record struct {
				Tool string `json:"tool"`
				Args any    `json:"args"`
			}
			decoder := json.NewDecoder(strings.NewReader(scanner.Text()))
			decoder.UseNumber()
			if err := decoder.Decode(&record); err != nil || record.Tool != "AX" {
				continue
			}
			args, ok := rewriteRefs(record.Args).(map[string]any)
			if !ok {
				args = map[string]any{}
			}
			accepted, text := historyVerdict(args)
			encoded, _ := json.Marshal(map[string]any{
				"file": path, "line": line, "accepted": accepted, "text": text,
			})
			writer.Write(append(encoded, '\n'))
		}
		file.Close()
	}
}
