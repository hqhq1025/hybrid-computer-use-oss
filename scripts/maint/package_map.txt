# File-to-package map for scripts/maint/split_packages.sh.
# "package <key> <dir> <name>" declares a destination; "<file-or-glob> <key>"
# assigns files. Exact names win over globs; the first matching glob wins.
# A _test.go file without its own rule follows its source file.
#
# Layering (lower packages never import higher ones):
#   bundle < contract < core < provider < transport < cmd
# core owns the platform port: the PlatformBackend interface and the
# Provider* request/result types it exchanges. provider implements the port
# (wire v2 encoding, Linux executor, darwin/windows sidecar transports).

package bundle    .                 runtimebundle
package contract  internal/contract contract
package core      internal/core     core
package provider  internal/provider provider
package transport internal/transport transport
package cmd       cmd/computer-use  main
